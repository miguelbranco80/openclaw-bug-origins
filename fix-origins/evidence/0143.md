+export function parseSessionEntries(content: string): FileEntry[] {
+  const entries: FileEntry[] = [];
+  let skipped = 0;
+  for (const line of content.trim().split("\n")) {
+    if (!line.trim()) {
+      continue;

+    try {

+        `${entries.length} valid entries were loaded`,
+    );
+  }
+  return entries;
