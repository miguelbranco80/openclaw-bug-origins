## What Problem This Solves

Fixes: automation run links show unrelated recent history when the requested run is older than the first 50 entries.

## User Impact

User impact: automation message and notification links retrieve the exact recorded run, with **Show all runs** to return to normal paginated history. Linked runs ignore leftover history filters; ordinary job selection keeps those filters.

## Why This Change Was Made

The existing history query now receives the linked identity and resolves it before pagination. Public run IDs retain precedence; session and execution aliases must identify one run within the applicable scope. Existing visibility checks remain in place. The current Solid controller initializes the history selection before publishing the editor, and the UI renders that query-owned selection without a second highlight state.

+export function resolveCronRouteData(search: string): {
+  jobId: string | null;
+  runId: string | null;
+} {
+  const params = new URLSearchParams(search);
+  const jobId = params.get("job")?.trim() || null;
+  return { jobId, runId: jobId ? params.get("run")?.trim() || null : null };
+}
+
+const CRON_EXECUTION_ID_RE = /^cron:(.+):(\d+)$/u;
+
+/**
+ * Notifications link runs by execution id (`cron:<jobId>:<startedAtMs>`), while
+ * ledger entries carry public run ids (receipt UUIDs, `manual:<...>` ids) that
+ * never equal it. Match exact ids first, then the entry's recorded run start.
+ */
+export function cronRunEntryMatchesLink(
+  linkedRunId: string,
+  entry: { jobId: string; runId?: string; runAtMs?: number },
+): boolean {
+  if (entry.runId === linkedRunId) {
+    return true;
+  }
+  const match = CRON_EXECUTION_ID_RE.exec(linkedRunId);
+  return match !== null && match[1] === entry.jobId && entry.runAtMs === Number(match[2]);
+}
