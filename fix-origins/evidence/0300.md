## Fix PR #202220 description (by steipete, opened 2026-11-20T03:36:36Z)

## What Problem This Solves

Opening an old chat with an unavailable Canvas widget can repeatedly request the same failed view whenever later chat events arrive or rows remount. Those requests also search the transcript before discovering that the widget file is gone, competing with other users' history and artifact reads.

## User Impact

Missing saved widgets keep their error visible until the page reloads or the agent creates a replacement. Temporary loading failures, including sandbox startup failures, retry at most five times with exponential backoff, then offer manual Retry. Successful widget responses are unchanged. No configuration or schema changes.

## Why This Change Was Made

The existing request owner now retains terminal failures across row remounts, retires their event subscriptions, and preserves retry backoff. Its request bookkeeping moves into a focused sibling module to stay within the view owner's line limit. Successful HTML remains owned only by mounted views.

The server opens a document immediately after session admission, before searching for its transcript receipt. Missing files release the retained read without entering the shared history-worker pool. Existing files still require the same receipt authorization before any content is returned. Both view responses and dashboard imports consume the document already read by that owner, so successful imports perform one source read. Sandbox startup failures are marked retryable after a valid source read; missing sources retain their terminal classification. Source loading and host startup remain parallel. An admitted session reader now receives generic UNAVAILABLE for a nonexistent arbitrary ID before receipt lookup, while present unreferenced IDs remain FORBIDDEN; document IDs are random UUIDs, and this reveals no bytes or paths.

+type WidgetClient = NonNullable<ApplicationContext["gateway"]["snapshot"]["client"]>;

+export type CanvasWidgetViewBinding = {

+export function isTerminalCanvasViewError(error: unknown): boolean {
+  return (
+    error instanceof GatewayProtocolRequestError &&
+    error.gatewayCode === "UNAVAILABLE" &&
+    !error.retryable
+  );
+}

+function canvasViewRequestKey(binding: CanvasWidgetViewBinding): string {
+  return `${binding.sessionKey}\0${binding.docId}\0${binding.operatorRead}`;
+}

+export function forgetPendingCanvasView(binding: CanvasWidgetViewBinding | undefined): void {
+  if (!binding) {
+    return;
+  }
+  const pending = pendingViews.get(binding.client);
+  if (pending?.generation === binding.generation) {
+    const key = canvasViewRequestKey(binding);
+    if (!pending.requests.get(key)?.terminal) {
+      pending.requests.delete(key);
+    }
+  }
+}

+export function loadCanvasView(
+  binding: CanvasWidgetViewBinding,
+): Promise<CanvasDocumentViewResult> {
+  let pending = pendingViews.get(binding.client);
+  if (!pending || pending.generation !== binding.generation) {
+    pending = { generation: binding.generation, requests: new Map() };
+    pendingViews.set(binding.client, pending);
+  }
+  const requestKey = canvasViewRequestKey(binding);
+  const existing = pending.requests.get(requestKey);
+  if (existing) {
+    return existing.result;
+  }
+  const request = binding.client.request<CanvasDocumentViewResult>(
+    "canvas.document.view",
+    {
+      docId: binding.docId,
+      ...(!binding.operatorRead && binding.sessionKey ? { sessionKey: binding.sessionKey } : {}),
+    },
+    { timeoutMs: WIDGET_LOAD_TIMEOUT_MS },
+  );
+  // Retain terminal misses through row remounts until reload/reconnection.
+  // Successful HTML stays with the mounted view, never in this request owner.
+  const requests = pending.requests;
+  const entry = { result: request, terminal: false };
+  requests.set(requestKey, entry);
+  void request
+    .catch((error: unknown) => {
+      entry.terminal = isTerminalCanvasViewError(error);
+    })
+    .finally(() => {
+      if (!entry.terminal && requests.get(requestKey) === entry) {
+        requests.delete(requestKey);
+      }
+    });
+  return request;
